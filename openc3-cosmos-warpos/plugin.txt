VARIABLE warplink_target WARPI

# Define the target (folder name in targets/ must match warplink_target)
TARGET <%= warplink_target %> <%= warplink_target %>

# Define the interface: uncomment the line that matches your hardware connection
#INTERFACE <%= warplink_target %>_INT openc3/interfaces/serial_interface.py /dev/ttyUSB0 /dev/ttyUSB0 115200 NONE 1 10.0 None # Serial to Linux
INTERFACE <%= warplink_target %>_INT openc3/interfaces/udp_interface.py 10.1.4.57 5006 5005 None None 128 10.0 None # RasPi UDP
#INTERFACE <%= warplink_target %>_INT openc3/interfaces/tcpip_client_interface.py host.docker.internal 5000 5000 10.0 None # Docker routing to Windows bridge
  # Fills the CRC item on every outgoing packet that has one. Skips packets
  # that don't (the bare-CFDP-PDU commands, e.g. STORAGE_MANAGER_CFDP -- their
  # own inner CFDP CRC is already baked into the raw PDU bytes before cmd()
  # is called, so there's no separate CRC item to fill). See
  # cfdp_aware_crc_protocol.py's own docstring for why the stock
  # crc_protocol.py can't be used unmodified here.
  PROTOCOL WRITE cfdp_aware_crc_protocol.py CRC False ERROR -16 16 BIG_ENDIAN 0x8005 0xFFFF False True

  # Delineates either an SSP packet or a CFDP PDU
  PROTOCOL READ check_pattern.py 512

  # Validates and removes the trailing CRC
  PROTOCOL READ openc3/interfaces/protocols/crc_protocol.py CRC False ERROR -16 16 BIG_ENDIAN 0x8005 0xFFFF False True

  MAP_TARGET <%= warplink_target %>

# CFDP file receiver microservice
MICROSERVICE CFDP_SERVICE CFDP_SERVICE
  CMD python3 cfdp_service.py
  TARGET_NAME <%= warplink_target %>
  OPTION TARGET <%= warplink_target %>
  OPTION PACKET CFDP_PACKET
  OPTION ITEM CFDP_PDU
  OPTION LOCAL_ENTITY_ID 0
  OPTION OUTPUT_DIR /openc3/user/cfdp
  OPTION COMMAND_TARGET <%= warplink_target %>
  OPTION COMMAND_PACKET STORAGE_MANAGER_CFDP
  OPTION COMMAND_ITEM data_buffer
  OPTION CFDP_BLOCK_SIZE 256
  OPTION OUTGOING_DIR /openc3/user/cfdp/outgoing
