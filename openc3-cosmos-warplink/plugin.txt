# plugin.txt - WarpLink plugin for WarpOS flight software
#
# One TARGET + INTERFACE block per WarpOS build target.  Each target needs its
# own UDP ports: COSMOS scopes packet identification to the targets mapped to
# the receiving interface, and the builds reuse the same STREAM_IDs.  Two
# targets sharing an interface — or two interfaces sharing a read port — cannot
# be told apart, and whichever interface the kernel hands a datagram to will
# claim it.
#
# Every connection setting is a VARIABLE, so it can be changed in the COSMOS
# plugin install dialog without editing this file.  Set <target>_enable to
# false to skip a target you are not flying.
#
# CosmosUpdateCmdTlm.py adds a block for each new build target it generates,
# on the next free ports, and never rewrites a block that already exists — so
# ports and hosts tuned here survive a regeneration.  To add one by hand, copy
# a block below and change the target name, the interface name, and the ports.

# ---------------------------------------------------------------------------
# SIM - simulation value injection
#
# Not a WarpOS build target: this is a send-only JSON channel used by the Sim
# Control tool to set simulation parameters, on its own port so it never mixes
# with flight software command traffic.  No read port - the simulation does not
# reply on this socket.
# ---------------------------------------------------------------------------
VARIABLE sim_enable            false
VARIABLE sim_host              host.docker.internal
VARIABLE sim_write_port        5009

# ---------------------------------------------------------------------------
# BF2_FLIGHT_BOARD
# ---------------------------------------------------------------------------
VARIABLE bf2_flight_board_enable      true
VARIABLE bf2_flight_board_host        10.1.4.27
VARIABLE bf2_flight_board_write_port  5006
VARIABLE bf2_flight_board_read_port   5005

# ---------------------------------------------------------------------------
# WARP_CUBE
# ---------------------------------------------------------------------------
VARIABLE warp_cube_enable      false
VARIABLE warp_cube_host        host.docker.internal
VARIABLE warp_cube_write_port  5011
VARIABLE warp_cube_read_port   5010

# ---------------------------------------------------------------------------
# NSNS
# ---------------------------------------------------------------------------
VARIABLE nsns_enable      false
VARIABLE nsns_host        host.docker.internal
VARIABLE nsns_write_port  5013
VARIABLE nsns_read_port   5012

# ---------------------------------------------------------------------------
# CFDP - file transfer
#
# CFDP_SERVICE reassembles files downlinked as bare CFDP PDUs (the
# CFDP_PACKET telemetry packet) into ./cfdp/incoming/complete/ on the host,
# and uplinks files queued by the CFDP Uplink tool through the
# STORAGE_MANAGER_CFDP command.  It serves one target: cfdp_target must be an
# enabled target whose cmd/tlm was generated with CFDP packets.
#
# Every target interface above uses cfdp_aware_crc_protocol.py (skips the CRC
# fill on bare CFDP PDU commands, which carry their own inner CRC) and
# check_pattern.py 512 (delineates both CCSDS packets and CFDP PDUs).
# ---------------------------------------------------------------------------
VARIABLE cfdp_enable            true
VARIABLE cfdp_target            BF2_FLIGHT_BOARD
VARIABLE cfdp_local_entity_id   0
VARIABLE cfdp_block_size        256

<% if sim_enable.to_s.strip.downcase == "true" %>
TARGET SIM SIM

# Write only UDP - read_port is None, so COSMOS never listens on this socket.
# Raw JSON datagrams: no CRC and no check_pattern protocols here on purpose.
INTERFACE SIM_INT openc3/interfaces/udp_interface.py <%= sim_host %> <%= sim_write_port %> None None None 128 10.0 None # Simulation JSON UDP
  MAP_TARGET SIM
<% end %>

<% if bf2_flight_board_enable.to_s.strip.downcase == "true" %>
TARGET BF2_FLIGHT_BOARD BF2_FLIGHT_BOARD

# Uncomment the line that matches your hardware connection
#INTERFACE BF2_FLIGHT_BOARD_INT openc3/interfaces/serial_interface.py /dev/ttyUSB0 /dev/ttyUSB0 115200 NONE 1 10.0 None # Serial to Linux
#INTERFACE BF2_FLIGHT_BOARD_INT openc3/interfaces/tcpip_client_interface.py host.docker.internal <%= bf2_flight_board_write_port %> <%= bf2_flight_board_read_port %> 10.0 None # Docker routing to Windows bridge
INTERFACE BF2_FLIGHT_BOARD_INT openc3/interfaces/udp_interface.py <%= bf2_flight_board_host %> <%= bf2_flight_board_write_port %> <%= bf2_flight_board_read_port %> None None 128 10.0 None # RasPi UDP
  PROTOCOL WRITE cfdp_aware_crc_protocol.py CRC False ERROR -16 16 BIG_ENDIAN 0x8005 0xFFFF False True
  PROTOCOL READ check_pattern.py 512
  PROTOCOL READ openc3/interfaces/protocols/crc_protocol.py CRC False ERROR -16 16 BIG_ENDIAN 0x8005 0xFFFF False True
  MAP_TARGET BF2_FLIGHT_BOARD
<% end %>

<% if cfdp_enable.to_s.strip.downcase == "true" %>
# CFDP file transfer microservice.  PACKET must match _CFDP_TLM_PACKET_NAME in
# CosmosUpdateCmdTlm.py; the output directories are the ./cfdp bind mount in
# compose.yaml (exposed to the CFDP Uplink tool as OPENC3_CFDP_VOLUME).
MICROSERVICE CFDP_SERVICE CFDP_SERVICE
  CMD python3 cfdp_service.py
  TARGET_NAME <%= cfdp_target %>
  OPTION TARGET <%= cfdp_target %>
  OPTION PACKET CFDP_PACKET
  OPTION ITEM CFDP_PDU
  OPTION LOCAL_ENTITY_ID <%= cfdp_local_entity_id %>
  OPTION OUTPUT_DIR /openc3/user/cfdp
  OPTION COMMAND_TARGET <%= cfdp_target %>
  OPTION COMMAND_PACKET STORAGE_MANAGER_CFDP
  OPTION COMMAND_ITEM data_buffer
  OPTION CFDP_BLOCK_SIZE <%= cfdp_block_size %>
  OPTION OUTGOING_DIR /openc3/user/cfdp/outgoing
<% end %>
