LANGUAGE python

# Not every packet in this plugin shares a uniform ID_ITEM layout (the CFDP
# telemetry packet identifies off multiple sub-byte bitfields in the CFDP
# header, unlike the 16-bit STREAM_ID used by the CCSDS-style packets) --
# without this, OpenC3's default hash-based identification reads every
# incoming packet's id values using only the first packet defined in this
# target, which silently breaks identification for any packet whose
# ID_ITEMs don't match that layout.
TLM_UNIQUE_ID_MODE

# Ignored Parameters
IGNORE_PARAMETER PKTID
IGNORE_PARAMETER STREAM_ID
IGNORE_PARAMETER PKT_LEN
IGNORE_PARAMETER SEQUENCE
IGNORE_PARAMETER SEC_HDR1
IGNORE_PARAMETER SEC_HDR2
IGNORE_PARAMETER SEC_HDR3
IGNORE_PARAMETER FILLER
IGNORE_PARAMETER CRC

# Ignored Items
IGNORE_ITEM PKTID
IGNORE_ITEM STREAM_ID
IGNORE_ITEM SEQUENCE
IGNORE_ITEM PKT_LEN
IGNORE_ITEM SEC_HDR1
IGNORE_ITEM SEC_HDR2
IGNORE_ITEM SEC_HDR3
IGNORE_ITEM FILLER
IGNORE_ITEM CRC
